git_clone_retry() {
    local url target attempt
    for arg in "$@"; do
        case "$arg" in
            http*|git@*|ssh://*|git://*) url="$arg" ;;
            -*) ;;
            *) [[ -n "$url" ]] && target="$arg" ;;
        esac
    done
    if [[ -z "$target" && -n "$url" ]]; then
        target="${url##*/}"
        target="${target%.git}"
    fi
    for attempt in 1 2 3 4 5; do
        if git clone "$@"; then
            return 0
        fi
        [[ -n "$target" && -d "$target" ]] && rm -rf "$target"
        local delay=$((attempt * 5))
        echo "git clone failed (attempt $attempt); retrying in ${delay}s..." >&2
        sleep "$delay"
    done
    echo "git clone failed after 5 attempts" >&2
    return 1
}

declare -A STAGED
declare -A SYSTEM_LIBS=(
    [libc.so.6]=1 [libm.so.6]=1 [libpthread.so.0]=1 [libdl.so.2]=1
    [librt.so.1]=1 [libresolv.so.2]=1 [libutil.so.1]=1
    [libstdc++.so.6]=1 [libgcc_s.so.1]=1
    [ld-linux-aarch64.so.1]=1
    [libSDL2-2.0.so.0]=1
)

resolve_lib_path() {
    # ldconfig -p line: "<soname> (<arch>) => <path>"
    ldconfig -p | awk -v lib="$1" '$1==lib {print $NF; exit}'
}

stage_lib_recursive() {
    local lib="$1"
    [[ -n "${STAGED[$lib]:-}" ]] && return 0
    [[ -n "${SYSTEM_LIBS[$lib]:-}" ]] && return 0
    STAGED[$lib]=1

    local src
    src=$(resolve_lib_path "$lib")
    if [[ -z "$src" || ! -f "$src" ]]; then
        echo "ERROR: cannot resolve $lib on build host (ldconfig miss)"; exit 1
    fi
    # cp -L follows the SONAME symlink chain to a real file; destination filename
    # matches the SONAME exactly so the loader resolves it by name.
    cp -L "$src" "../libs/$lib"
    echo "  staged: $lib  ($src)"

    local dep
    while read -r dep; do
        [[ -z "$dep" ]] && continue
        stage_lib_recursive "$dep"
    done < <(readelf -d "$src" 2>/dev/null | awk -F'[][]' '/NEEDED/ {print $2}')
}

needed=$(readelf -d zelda3 | awk -F'[][]' '/NEEDED/ {print $2}')
echo "zelda3 NEEDED:"; echo "$needed"
echo "Staging transitive lib closure into ../libs/ ..."
while read -r lib; do
    [[ -z "$lib" ]] && continue
    stage_lib_recursive "$lib"
done <<< "$needed"

echo "Final ../libs/ contents:"; ls -la ../libs/
echo "zelda3 max required GLIBC symbol:"
readelf -V zelda3 2>/dev/null | grep -oE 'GLIBC_[0-9]+\.[0-9]+' | sort -V | tail -1 || true

cp zelda3 ../zelda3
cd ..
