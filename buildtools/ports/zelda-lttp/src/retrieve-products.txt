#!/bin/bash
set -e

SRCDIR=$1
DESTDIR=$2

cd "$DESTDIR" || { echo "DESTDIR not found: $DESTDIR"; exit 1; }

# Binary
cp "$SRCDIR/zelda3" "$DESTDIR/zelda3"
chmod +x "$DESTDIR/zelda3"

# Bundled libs
rm -rf "$DESTDIR/libs"
mkdir -p "$DESTDIR/libs"
cp -a "$SRCDIR/libs/." "$DESTDIR/libs/" 2>/dev/null || true

echo "Packaging complete."
